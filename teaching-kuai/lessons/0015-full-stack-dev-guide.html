Step 1：数据库建表

设计评估任务表
-- 评估任务表
CREATE TABLE `eval_task` (
  `id`            BIGINT AUTO_INCREMENT PRIMARY KEY,
  `name`          VARCHAR(128) NOT NULL COMMENT '任务名称',
  `prompt_id`     BIGINT NOT NULL COMMENT '关联 Prompt ID',
  `prompt_version` INT NOT NULL COMMENT 'Prompt 版本号',
  `dataset_id`    BIGINT NOT NULL COMMENT '测试数据集 ID',
  `model`         VARCHAR(64) NOT NULL COMMENT 'LLM 模型名',
  `status`        TINYINT DEFAULT 0 COMMENT '0=待运行 1=运行中 2=完成 3=失败',
  `result_json`   TEXT COMMENT '评估结果 JSON',
  `create_time`   DATETIME DEFAULT CURRENT_TIMESTAMP,
  `update_time`   DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  INDEX `idx_prompt_id` (`prompt_id`),
  INDEX `idx_status` (`status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='评估任务表';

Service 实现
@Service
@Slf4j
public class EvalTaskServiceImpl extends ServiceImpl<EvalTaskMapper, EvalTaskEntity>
        implements EvalTaskService {

    @Override
    @Transactional(rollbackFor = Exception.class)
    public EvalTaskDTO create(EvalTaskCreateReq req) {
        EvalTaskEntity entity = new EvalTaskEntity();
        BeanUtils.copyProperties(req, entity);
        entity.setStatus(0); // 待运行
        save(entity);
        log.info("创建评估任务: id={}, name={}", entity.getId(), entity.getName());
        return toDTO(entity);
    }

    @Override
    public EvalTaskDTO run(Long taskId) {
        EvalTaskEntity task = getById(taskId);
        if (task == null) throw new BizException("任务不存在");

        task.setStatus(1); // 运行中
        updateById(task);

        try {
            // 调用评估引擎运行评估
            String result = evaluationEngine.run(task);
            task.setResultJson(result);
            task.setStatus(2); // 完成
        } catch (Exception e) {
            task.setStatus(3); // 失败
            log.error("评估任务执行失败: id={}", taskId, e);
        }

        updateById(task);
        return toDTO(task);
    }
}
Controller 接口
@RestController
@RequestMapping("/api/eval-task")
@Slf4j
public class EvalTaskController {

    @Autowired
    private EvalTaskService evalTaskService;

    @PostMapping("/create")
    public Result<EvalTaskDTO> create(@RequestBody @Valid EvalTaskCreateReq req) {
        return Result.success(evalTaskService.create(req));
    }

    @PostMapping("/{id}/run")
    public Result<EvalTaskDTO> run(@PathVariable Long id) {
        return Result.success(evalTaskService.run(id));
    }

    @GetMapping("/list")
    public Result<List<EvalTaskDTO>> list(
            @RequestParam(required = false) Integer status,
            @RequestParam(defaultValue = "1") Integer page,
            @RequestParam(defaultValue = "10") Integer size) {
        return Result.success(evalTaskService.listByStatus(status, page, size));
    }

    @GetMapping("/{id}")
    public Result<EvalTaskDTO> detail(@PathVariable Long id) {
        return Result.success(evalTaskService.getById(id));
    }
}

Step 3：前端页面开发

页面组件结构
// app/eval-tasks/page.tsx
'use client';
import { useState, useEffect } from 'react';
import { Table, Button, Modal, Form, Select, message } from 'antd';

export default function EvalTasksPage() {
  const [tasks, setTasks] = useState([]);
  const [loading, setLoading] = useState(false);

  // 加载任务列表
  const fetchTasks = async () => {
    setLoading(true);
    const res = await fetch('/api/eval-task/list');
    const { data } = await res.json();
    setTasks(data);
    setLoading(false);
  };

  useEffect(() => { fetchTasks(); }, []);

  // 创建评估任务
  const handleCreate = async (values) => {
    const res = await fetch('/api/eval-task/create', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(values),
    });
    const { code, message: msg } = await res.json();
    if (code === 200) {
      message.success('创建成功');
      fetchTasks();
    }
  };

  // 运行评估任务
  const handleRun = async (id: number) => {
    await fetch(`/api/eval-task/${id}/run`, { method: 'POST' });
    message.success('任务已提交');
    fetchTasks();
  };

  const columns = [
    { title: '任务名称', dataIndex: 'name' },
    { title: '模型', dataIndex: 'model' },
    {
      title: '状态', dataIndex: 'status',
      render: (s) => ['待运行', '运行中', '完成', '失败'][s]
    },
    {
      title: '操作',
      render: (_, record) => (
        <>
          <Button onClick={() => handleRun(record.id)}>运行</Button>
          <Button href={`/eval-tasks/${record.id}`}>查看</Button>
        </>
      ),
    },
  ];

  return (
    <div>
      <Button type="primary" onClick={() => setVisible(true)}>新建任务</Button>
      <Table dataSource={tasks} columns={columns} loading={loading} />
    </div>
  );
}

3.1 日志规范

级别用途示例
ERROR 影响功能的错误 数据库连接失败、API 调用异常
WARN 潜在问题 重试次数接近上限、慢查询
INFO 关键业务事件 用户登录、任务创建、评估完成

3.3 前端监控

// 前端错误监控
window.addEventListener('unhandledrejection', (event) => {
  console.error('Unhandled promise rejection:', event.reason);
  // 上报到监控系统
  reportError({ type: 'unhandledrejection', error: event.reason });
});

// API 请求错误拦截
const originalFetch = window.fetch;
window.fetch = async (...args) => {
  try {
    const response = await originalFetch(...args);
    if (!response.ok) {
      console.error(`API Error: ${args[0]} -> ${response.status}`);
    }
    return response;
  } catch (error) {
    console.error(`Network Error: ${args[0]}`, error);
    throw error;
  }
};

3.4 关键监控指标

指标说明告警阈值
API 响应时间P99 延迟> 3s

评估任务成功率成功完成的任务占比< 95%
数据库连接池活跃连接数 / 最大连接数> 80%

生产环境建议：
  • 使用结构化日志（JSON 格式），便于日志系统采集和检索
  • 为每个请求生成 Trace ID，串联前后端日志
  • 设置日志轮转，避免磁盘打满
  • 关键业务指标接入 Grafana 看板

📝 Quiz

Q1：在 Kuai 平台的全栈开发中，新增一个功能的标准开发顺序是？

  • A. 前端页面 → 后端接口 → 数据库建表
  • B. 数据库建表 → 后端接口 → 前端页面 → 联调测试
  • C. 后端接口 → 前端页面 → 数据库建表
  • D. 同时开发前端和后端，最后建表

✅ 答案：B — 标准流程是先设计数据模型（建表），再实现后端逻辑（API），然后开发前端界面，最后联调验证。

Q2：生产环境部署时，数据库密码等敏感信息应该如何管理？

  • A. 直接写在 application.yml 文件中
  • B. 写在代码注释里方便查看
  • C. 使用环境变量或密钥管理服务，不提交到代码仓库
  • D. 通过前端页面传给后端

✅ 答案：C — 敏感信息应通过环境变量（${DB_PASS}）或密钥管理服务注入，绝不能硬编码或提交到 Git。

Q3：日志级别中，「评估任务执行失败」应该使用哪个级别？

  • A. DEBUG — 便于开发调试
  • B. INFO — 记录关键事件
  • C. WARN — 潜在问题
  • D. ERROR — 影响功能的错误

✅ 答案：D. ERROR — 任务执行失败影响核心功能，且通常伴随异常堆栈，应使用 ERROR 级别并记录完整异常信息。

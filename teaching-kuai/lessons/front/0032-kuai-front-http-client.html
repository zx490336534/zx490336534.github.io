🔧 request 请求封装详解

核心文件 src/services/request/index.js 是整个前端的 HTTP 请求基础，基于浏览器原生 fetch API 进行封装。

核心设计：直接返回数据

// src/services/request/index.js (简化)
export default function request(url, options, style) {
  // 兼容旧版调用方式：request(url, 'json')
  let actualOptions = options;
  let actualResponseType = style;

  if (typeof options === 'string') {
    actualOptions = {};
    actualResponseType = options;
  } else if (!options) {
    actualOptions = {};
  }

  const finalResponseType = actualResponseType || RESPONSE_TYPE.JSON;

  const defaultOptions = {
    credentials: 'include',  // 自动携带 Cookie（SSO 认证）
  };

  const newOptions = { ...defaultOptions, ...actualOptions };
  setHeadersForBody(newOptions);  // 自动设置 Content-Type

  // 直接返回接口数据
  return fetch(url, newOptions)
    .then(checkStatus)           // 检查状态码
    .then(response => parseResponse(response, finalResponseType));
}

💡 关键设计决策

request 函数直接返回解析后的数据，而不是包装成 { data, err }。错误处理通过 Promise 的 reject（throw）机制完成。这意味着调用方需要使用 try/catch 或 .catch() 来捕获错误。

请求流程

组件调用 API 函数
↓
request(url, options) — 合并默认配置
↓
setHeadersForBody() — 自动设置 Content-Type / JSON 序列化
↓
fetch(url, options) — 发起 HTTP 请求（credentials: 'include'）
↓
checkStatus(response) — 检查状态码，401 跳转 SSO
↓
parseResponse() — 根据 responseType 解析（JSON/Text/Blob）
↓
返回解析后的数据

三种响应类型

类型常量用途示例
JSONRESPONSE_TYPE.JSON默认，API 返回 JSON 数据用户信息、列表数据
TextRESPONSE_TYPE.TEXT返回纯文本流式响应、日志
BlobRESPONSE_TYPE.BLOB返回二进制数据文件下载、图片
// 使用不同的响应类型
import request, { RESPONSE_TYPE } from '@/services/request';

// 默认 JSON
const userData = await request('/api/user/current');

// 纯文本
const logText = await request('/api/logs', {}, RESPONSE_TYPE.TEXT);

// 二进制文件
const fileBlob = await request('/api/export', {}, RESPONSE_TYPE.BLOB);

⚠️ 统一错误处理

HTTP 状态码处理

checkStatus 函数是错误处理的核心，它对每种 HTTP 状态码都有对应的处理逻辑：

状态码处理方式说明
200-299正常返回成功响应，继续解析数据
401自动跳转 SSO 登录认证失败，重定向到登录页
403抛出错误无权限访问
404抛出错误接口不存在
500+抛出错误服务器内部错误
// src/services/request/index.js — checkStatus 核心逻辑
function checkStatus(response) {
  // 401 未授权：自动跳转 SSO 登录
  if (response.status === 401) {
    if (isGuestMode()) {
      return response;  // 访客模式不跳转
    }
    redirectToLogin();  // 跳转到 SSO 登录页
    throw new Error('Unauthorized - redirecting to login');
  }

  // 成功响应 (2xx)
  if (response.status >= 200 && response.status < 300) {
    return response;
  }

  // 其他错误
  const errorMessage = STATUS_MESSAGES[response.status] || response.statusText;
  const error = new Error(errorMessage);
  error.name = response.status;
  error.response = response;
  throw error;
}

访客模式

🔑 访客模式（免登录）

当 URL 同时满足以下两个条件时，进入访客模式：

  • 路径以 /share 开头（分享页面）
  • URL 参数包含 ?mode=guest

访客模式下，401 错误不会触发跳转，而是直接返回响应，让页面自行处理。

JSON 解析的容错处理

// 解析响应数据，支持大数字和容错
async function parseResponse(response, responseType) {
  switch (responseType) {
    case RESPONSE_TYPE.JSON: {
      const clonedResponse = response.clone();
      const text = await clonedResponse.text();
      if (!text || text.trim().length === 0) {
        return null;  // 空响应返回 null
      }
      try {
        return JSONBig.parse(text);  // 使用 json-bigint 处理大数字
      } catch (err) {
        console.warn('[request] JSON parse failed, returning text');
        return text;  // 解析失败返回原始文本
      }
    }
    case RESPONSE_TYPE.TEXT:
      return response.text();
    case RESPONSE_TYPE.BLOB:
      return response.blob();
  }
}

🔍 为什么用 JSONBig？

JavaScript 原生 JSON.parse 无法正确处理超过 Number.MAX_SAFE_INTEGER（2^53 - 1）的整数。后端返回的 ID 或时间戳可能是大数字，使用 json-bigint 库可以避免精度丢失。

📦 模块化 API 服务

每个业务模块在 src/services/modules/ 下有自己的 API 文件，遵循统一的编码规范：

模块编写规范

import request from '@/services';

// 1. 定义 API 端点常量

  return request(API_USER_LOGOUT);

export async function queryStaticUserMap() {
  return request(API_USER_STATIC_MAP);

🎯 实际使用模式

在页面组件中调用 API

// 页面组件中使用 API 的典型模式
'use client';
import { useState, useEffect } from 'react';
import { getServers } from '@/services';

export default function McpServersPage() {
  const [servers, setServers] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function loadServers() {
      try {
        const data = await getServers();  // 直接获取数据
        setServers(data || []);
      } catch (err) {
        console.error('加载服务器列表失败:', err);
        // request 内部已处理 401 跳转，这里处理其他错误
      } finally {
        setLoading(false);
      }
    }
    loadServers();
  }, []);

  if (loading) return <div>加载中...</div>;

  return (
    <div>
      {servers.map(server => (
        <div key={server.name}>{server.name}</div>
      ))}
    </div>
  );
}

不同后端的 API 代理

🔀 API 代理规则

不同模块的 API 前缀指向不同的后端服务：

  • /api/user/* → Kuai 后端（用户认证）
  • /api/proxy/mcphub/* → Python MCP 服务（MCP 管理）
  • /ai/api/* → AI 后端（提示词管理）
  • /api/moon/* → Moon 服务
  • /kaptain/api/* → Kaptain 服务

这些代理规则在 next.config.js 的 rewrites 中配置（详见 0024 课程）。

请求头自动处理

// setHeadersForBody 自动处理请求头
function setHeadersForBody(options) {
  const { method, body, headers } = options;

  // 只对有请求体的方法设置头部
  if (!['POST', 'PUT', 'DELETE'].includes(method) || !body) {
    return;
  }

  if (body instanceof FormData) {
    // 文件上传：不设置 Content-Type（浏览器自动设置 boundary）
    options.headers = { Accept: 'application/json', ...headers };
  } else {
    // 普通 JSON 请求
    options.headers = {
      Accept: 'application/json',
      'Content-Type': 'application/json; charset=utf-8',
      ...headers,
    };
    options.body = JSONBig.stringify(body);  // JSON 序列化
  }
}

🧪 知识检验

题目 1：请求封装

当后端返回 HTTP 401 状态码时，request 函数会如何处理？

A. 抛出错误，让调用方自行处理
B. 返回 null 数据
C. 自动重定向到 SSO 登录页面（访客模式除外）
D. 自动刷新当前页面
答案 C。checkStatus 函数检测到 401 状态码后，会先判断是否处于访客模式（/share?mode=guest）。如果不是访客模式，调用 redirectToLogin() 将用户重定向到 SSO 登录页。这是统一的认证处理机制，所有页面都不需要单独处理 401。

题目 2：模块组织

为什么 src/services/modules/mcp.js 中的 API 端点使用函数而不是字符串常量？

A. 为了代码好看
B. 因为端点包含动态参数（如服务器名称），需要用函数拼接路径
C. 因为 JavaScript 不支持字符串模板
D. 这是团队的代码规范，没有实际意义
答案 B。MCP 模块的很多端点包含动态参数，如 /api/proxy/mcphub/servers/${name}。使用函数可以安全地拼接路径：const SERVER_BY_NAME = (name) => `/api/proxy/mcphub/servers/${name}`。调用时 request(SERVER_BY_NAME('my-server'))。对于没有动态参数的端点（如列表接口），则直接使用字符串常量。

题目 3：统一入口

src/services/index.js 文件的作用是什么？

A. 定义所有 API 的具体实现
B. 配置 HTTP 请求的超时时间和重试策略
C. 作为统一入口，重新导出 request 函数和所有模块的 API
D. 处理所有 API 的请求拦截和响应拦截
答案 C。src/services/index.js 是服务层的统一入口文件。它做两件事：(1) 默认导出 request 函数；(2) 通过 export * from './modules/xxx' 重新导出所有模块的 API 函数。这样页面组件只需 import { getServers } from '@/services' 就能使用任何 API，不需要记住具体模块路径。

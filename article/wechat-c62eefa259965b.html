数据库设计

设备ip device_ip唯一

  1. from django.db import models
  2. from utils.base_models import BaseModel
  4. status_choice = ((0, '空闲'), (1, '使用中'))
  5. sd_status_choice = ((0, '无'), (1, '有'))
  8. class Device_info(BaseModel):
  9.     id = models.AutoField(verbose_name='id主键', primary_key=True, help_text='id主键')
  10.     device_ip = models.CharField('设备IP', max_length=200, unique=True, help_text='设备IP')
  11.     device_name = models.CharField('设备名称', max_length=200, help_text='设备名称')
  12.     device_version = models.CharField('设备版本', max_length=200, help_text='版本信息')
  13.     tool_ip = models.CharField('工装设备IP', max_length=200, null=True, blank=True, default='', help_text='工装设备IP')
  14.     tool_pc_ip = models.CharField('工装电脑IP', max_length=200, null=True, blank=True, default='', help_text='工装电脑IP')
  15.     daily_status = models.IntegerField('每日编译状态', choices=status_choice, default=0, help_text='每日编译状态')
  16.     use_status = models.IntegerField('设备使用状态', choices=status_choice, default=0, help_text='设备使用状态')
  17.     sd_status = models.IntegerField('SD卡状态', choices=sd_status_choice, default=0, help_text='SD卡状态')
  18.     address = models.CharField('设备位置', max_length=200, null=True, blank=True, default='', help_text='设备位置')
  20.     class Meta:
  21.         db_table = 'u_device_info'
  22.         verbose_name = '设备信息'
  23.         verbose_name_plural = verbose_name
  25.     def __str__(self):
  26.         return f"{self.device_ip}-{self.device_name}"

全部代码

  1. @action(methods=['post'], detail=False)
  2. def status(self, request):
  3.     """
  4.     获取设备状态
  5.     {
  6.         "device_ip":"",
  7.         "device_name":"",
  8.         "type":"daily/tool/unuse"
  9.     }
  10.     """
  11.     data = request.data
  12.     device_ip = data.get('device_ip')
  13.     device_name = data.get('device_name')
  14.     type = data.get('type')
  15.     device_info_objs = self.queryset
  16.     if device_ip:
  17.         # 根据设备IP查找,返回一个
  18.         device_info_objs = self.queryset.filter(device_ip=device_ip).first()
  19.     elif device_name:
  20.         # 根据设备名称查找,返回全部
  21.         device_info_objs = self.queryset.filter(device_name=device_name).all()
  22.     elif type:
  23.         if type == 'daily':
  24.             # 每日编译状态为0 且设备使用状态为0,返回一个
  25.             device_info_objs = self.queryset.filter(daily_status=0, use_status=0).first()
  26.         elif type == 'tool':
  27.             # 有工装IP 同一工装电脑下设备使用状态为0,返回全部
  28.             # 查找 tool_pc_ip有「.」内容且 use_status为1的全部电脑IP
  29.             working = self.queryset.values('tool_pc_ip').annotate(use=Count('use_status')).filter(
  30.                 use_status=1).filter(tool_pc_ip__contains='.').all()
  31.             working_list = list(working)
  32.             if working_list:
  33.                 print([f"跳过的工装PC:{i.get('tool_pc_ip')}" for i in working_list])
  34.                 device_info_objs = self.queryset.filter(~Q(tool_pc_ip=working_list[0].get('tool_pc_ip')))
  35.                 device_info_objs = device_info_objs.filter(tool_pc_ip__contains='.')
  36.                 for i in working_list[1:]:
  37.                     a = ~Q(tool_pc_ip=i.get('tool_pc_ip'))
  38.                     device_info_objs = device_info_objs.filter(a)
  39.             else:
  40.                 return Response({'status': -1, 'error': '没有空闲的工装设备'})
  41.         elif type == 'unuse':
  42.             # 任意非使用状态的普通设备,返回全部
  43.             device_info_objs = self.queryset.filter(is_delete=False, use_status=0).filter(tool_pc_ip='').filter(
  44.                 tool_ip='').all()
  45.     else:
  46.         return Response({'status': -1, 'error': '请使用正确的查询方式'})
  47.     if device_ip or type == 'daily':
  48.         serializer = self.get_serializer(device_info_objs, many=False)
  49.     else:
  50.         serializer = self.get_serializer(device_info_objs, many=True)
  51.     new_data = get_count_by_device_info(serializer.data)
  52.     return Response(new_data)

修改设备状态

从数据库中检索 device_ip=device_ip的内容

  1. device_info = self.queryset.filter(device_ip=device_ip).first()
  2. device_info.use_status = use_status
  3. device_info.daily_status = daily_status
  1. @action(methods=['post'], detail=False)
  2. def use(self, request, *args, **kwargs):
  3.    """
  4.    修改设备使用状态
  5.         {
  6.             "device_ip":'',
  7.             "use_status":0或1,
  8.             "daily_status":0或1
  9.         }
  10.     """
  11.     data = request.data
  12.     device_ip = data.get('device_ip')
  13.     use_status = data.get('use_status')
  14.     daily_status = data.get('daily_status')
  15.     device_info = self.queryset.filter(device_ip=device_ip).first()
  16.     if use_status in [0, 1] and daily_status in [0, 1] and device_info:
  17.       device_info.use_status = use_status
  18.       device_info.save()
  19.       serializer = serializers.DeviceModelSerializer(instance=device_info)
  20.       return Response(serializer.data)
  21.     else:
  22.       return Response({'status': -1, 'error': 'use_status和daily_status只能为0或1'})
